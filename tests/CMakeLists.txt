# Also usable by itself: cmake -S tests -B build/area-tests
cmake_minimum_required(VERSION 3.23)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(LocationAreaTests LANGUAGES CXX)
    enable_testing()
endif()

add_executable(LocationAreaTests AreasTests.cpp)
target_compile_features(LocationAreaTests PRIVATE cxx_std_20)
target_include_directories(LocationAreaTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../src")
if(MSVC)
    target_compile_options(LocationAreaTests PRIVATE /utf-8 /W4)
endif()
add_test(NAME LocationAreaTests COMMAND LocationAreaTests)

add_executable(NotificationTests NotificationsTests.cpp)
target_compile_features(NotificationTests PRIVATE cxx_std_20)
target_include_directories(NotificationTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../src")
if(MSVC)
    target_compile_options(NotificationTests PRIVATE /utf-8 /W4)
endif()
add_test(NAME NotificationTests COMMAND NotificationTests)

add_executable(DistrictTests DistrictsTests.cpp)
target_compile_features(DistrictTests PRIVATE cxx_std_20)
target_include_directories(DistrictTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../src")
if(MSVC)
    target_compile_options(DistrictTests PRIVATE /utf-8 /W4)
endif()
add_test(NAME DistrictTests COMMAND DistrictTests)

# Explicit opt-in: test companion only, never copied or packaged with the mod.
option(DLP_BUILD_LAB_PROBE "Build private-desktop in-game verification companion" OFF)
if(DLP_BUILD_LAB_PROBE)
    find_package(fmt REQUIRED CONFIG)
    find_package(spdlog REQUIRED CONFIG)
    add_library(DLPVerificationProbe SHARED LabProbe.cpp)
    target_compile_features(DLPVerificationProbe PRIVATE cxx_std_23)
    target_include_directories(DLPVerificationProbe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../src")
    target_precompile_headers(DLPVerificationProbe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../src/PCH.h")
    target_link_libraries(DLPVerificationProbe PRIVATE CommonLibF4::CommonLibF4 fmt::fmt spdlog::spdlog d3d11 dxgi)
    target_compile_options(DLPVerificationProbe PRIVATE /EHa /utf-8 /Zc:preprocessor /permissive-)
endif()
